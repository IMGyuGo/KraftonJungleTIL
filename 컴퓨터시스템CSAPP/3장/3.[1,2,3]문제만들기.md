```c
long mystery(long *arr, long n, long limit) {
    long i = 0;
    long sum = 0;

    while (i < n) {
        long val = arr[i];

        if (val > limit) {
            sum += val;
        } else {
            sum -= val;
        }

        i++;
    }

    return sum;
}
```

```
    subq    B, %rcx
```
