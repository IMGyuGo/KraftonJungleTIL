/*
 * mm_malloc - brk 포인터를 늘려 블록 할당.
 *     항상 정렬 배수 크기의 블록을 할당한다.
 */
void *mm_malloc(size_t size)
{

    // 8바이트 (헤드랑 푸터를 더해서 사이즈를 할당)
    int newsize = ALIGN(size + SIZE_T_SIZE);
    // 새로운 사이즈를 할당하고, p에 할당 이전 주소만 반환
    void *p = mem_sbrk(newsize);
    if (p == (void *)-1)
        return NULL;
    else
    {
        // 오류가 안발생했으면, p의 8바이트 사이즈 안에 사이즈 정보 추가
        *(size_t *)p = size;

        // 마지막 리턴 값은 p의 NEXT HDRP의 주소
        return (void *)((char *)p + SIZE_T_SIZE);
    }

    /*
    malloc을 하는 순간
    *p = mem_brk 현재 블록의 시작 주소
    *(size_t *)p = mem_brk 현재 블록의 시작 주소에 payload 크기 저장
    마지막 리턴 값은 = p의 NEXT HDRP의 주소
    */
}
